💻 Access Control Algorithm Implementation

# Tree Node Data Structure class AccessTreeNode: def __init__(self, user_id, name, role, parent=None): self.user_id = user_id self.name = name self.role = role # GM, PM, TL, MEMBER self.parent = parent self.children = [] self.permissions = self._resolve_permissions() self.connection_methods = self._resolve_connection_methods() def _resolve_permissions(self): """Dynamically resolve FTP directory permissions based on role""" permission_map = { 'GM': ['/all_directories'], 'PM': [ '/PM_personal/' + self.user_id, '/PM_share/', '/techlead_share/', '/shared_access/' ], 'TL': [ '/TL_personal/' + self.user_id, '/TL_share/', '/shared_access/', '/Employee_personal/' + '*' # Restricted to direct reports ], 'MEMBER': [ '/Employee_personal/' + self.user_id ] } return permission_map.get(self.role, []) def _resolve_connection_methods(self): """Resolve allowed connection methods based on role""" if self.role in ['GM', 'PM', 'TL']: return {'VPN_88': True, 'VDI': True, 'LOCAL': True} else: return {'VPN_88': True, 'VDI': False, 'LOCAL': True} # Tree Traversal Algorithm class AccessControlTree: def __init__(self): self.root = None self.user_index = {} # Fast lookup by user_id def build_tree(self, hierarchy_data): """Build tree from hierarchical JSON data""" # Implementation creates nodes and establishes parent-child relationships pass def find_user(self, user_id): """O(1) lookup using hash map index""" return self.user_index.get(user_id) def get_direct_reports(self, tl_user_id): """Get all direct reports for a Tech Lead""" tl_node = self.find_user(tl_user_id) if tl_node and tl_node.role == 'TL': return [child for child in tl_node.children] return [] def resolve_user_access(self, user_id): """Main algorithm: Resolve complete access profile for a user""" user_node = self.find_user(user_id) if not user_node: raise ValueError(f"User {user_id} not found") # Step 1: Get base permissions from role base_permissions = user_node.permissions # Step 2: For TLs, restrict Employee_personal to direct reports only if user_node.role == 'TL': direct_reports = self.get_direct_reports(user_id) employee_paths = [ f'/Employee_personal/{report.user_id}' for report in direct_reports ] # Replace wildcard with specific paths base_permissions = [ p for p in base_permissions if '*' not in p ] + employee_paths # Step 3: Get connection methods connection_methods = user_node.connection_methods # Step 4: Build access profile access_profile = { 'user_id': user_node.user_id, 'name': user_node.name, 'role': user_node.role, 'ftp_directories': base_permissions, 'connection_methods': connection_methods, 'path_to_root': self._get_path_to_root(user_node) } return access_profile def _get_path_to_root(self, node): """Trace path from node to root (for audit trail)""" path = [] current = node while current: path.append(current.user_id) current = current.parent return list(reversed(path)) #   • Active Directory Integration: Map AD groups to roles (GM, PM, TL, MEMBER) and sync the organizational hierarchy automatically.
  • Virtual Paths: Configure virtual paths in Wing FTP to map logical directory structures to physical storage locations.
